-- Month-over-month user retention WITH prior_month AS ( SELECT DISTINCT user_id FROM access_log WHERE DATE_FORMAT(created_at, '%Y-%m') = '2025-04' ), current_month AS ( SELECT DISTINCT user_id FROM access_log WHERE DATE_FORMAT(created_at, '%Y-%m') = '2025-05' ) SELECT COUNT(DISTINCT p.user_id) AS retained_users FROM prior_month p JOIN current_month c ON p.user_id = c.user_id;
✓ retention cohort resolvedused in every QBR deck
-- Unique active users bucketed by hour of day SET SESSION MAX_EXECUTION_TIME = 600000; SELECT HOUR(created_at) AS activity_hour, COUNT(DISTINCT user_id) AS hourly_dau FROM access_log WHERE created_at BETWEEN '2025-02-01' AND '2025-03-01' GROUP BY HOUR(created_at) ORDER BY activity_hour;

-- Full account overview: trunks, DIDs, agents, limits SELECT u.login_id, u.client_id, u.company_name, p.product_name, p.amount_paid, p.allowed_agents, dids.did_count, trunks.c2c_trunk, trunks.dialer_trunk, trunks.agent_trunk, CASE WHEN p.deleted_at IS NOT NULL THEN 'closed' ELSE 'active' END AS status FROM billing_user_products p LEFT JOIN users u ON p.user_id = u.id AND u.is_deleted = 0 LEFT JOIN did_agg dids ON dids.client_id = u.client_id LEFT JOIN trunk_agg trunks ON trunks.client_id = u.client_id WHERE u.client_id IS NOT NULL AND p.deleted_at IS NULL;

-- CRM integration adoption by client SELECT c.crm_name, COUNT(DISTINCT i.client_id) AS clients, COUNT(DISTINCT i.user_id) AS active_users, ROUND(COUNT(DISTINCT i.user_id) / NULLIF(COUNT(DISTINCT i.client_id),0), 1) AS users_per_client FROM integrations i JOIN crm_catalog c ON c.id = i.crm_id WHERE i.is_active = 1 GROUP BY c.crm_name ORDER BY active_users DESC;
